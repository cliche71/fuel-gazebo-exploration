cmake_minimum_required(VERSION 3.0.2)
project(fuel_camera_adapter)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs std_msgs gazebo_msgs)
find_package(gazebo REQUIRED)
catkin_package(LIBRARIES fuel_depth_camera fuel_contact_observer)
include_directories(${catkin_INCLUDE_DIRS} ${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
set(CMAKE_CXX_STANDARD 14)
add_library(fuel_depth_camera SHARED src/depth_camera.cpp)
target_link_libraries(fuel_depth_camera ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES} DepthCameraPlugin)
add_library(fuel_contact_observer SHARED src/contact_observer.cpp)
target_link_libraries(fuel_contact_observer ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES})
install(TARGETS fuel_depth_camera fuel_contact_observer LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})

add_library(fuel_seed_evidence SHARED src/seed_observer.cpp)
target_link_libraries(fuel_seed_evidence ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES})
